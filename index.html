async function cargarArchivosAutomaticos() { document.getElementById('infoContador').textContent = 'Cargando archivos CSV desde la nube...'; try { // Carga de pendientes const respPend = await fetch('./pendientes.csv'); if (!respPend.ok) throw new Error('No se pudo abrir pendientes.csv'); const textoPend = await respPend.text(); const resPend = parsearCSV(textoPend); cabeceras = resPend.cabeceras; datosPendientes = resPend.filas; if (!cabeceras.includes('Inventario Disponible en el punto de origen')) { cabeceras.push('Inventario Disponible en el punto de origen'); } if (!cabeceras.includes('Inventario BODEGA PRINCIPAL CEDI')) { cabeceras.push('Inventario BODEGA PRINCIPAL CEDI'); } // Carga de inventario const respInv = await fetch('./ValorizadoXls2202698125353.csv'); if (respInv.ok) { const textoInv = await respInv.text(); const resInv = parsearCSV(textoInv); procesarCruzeInventario(resInv.filas); } guardarEnLocalStorage(); poblarSelectoresFiltro(); aplicarFiltros(); } catch (err) { document.getElementById('infoContador').textContent = 'Error al cargar datos automáticos. Use los botones para subir CSV.'; console.error(err); } }
