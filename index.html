async function cargarArchivosAutomaticos() { document.getElementById('infoContador').textContent = 'Cargando archivos CSV desde la nube...'; try { const timestamp = `?v=${new Date().getTime()}`; // 1. Carga del archivo de pendientes con codificación ISO-8859-1 y lectura de espacios en la URL const archivoPendientes = 'Total_Dispensacion_Pendientes Activos.csv'; const respPend = await fetch(`./${encodeURIComponent(archivoPendientes)}${timestamp}`); if (!respPend.ok) throw new Error(`No se pudo abrir ${archivoPendientes}`); const bufferPend = await respPend.arrayBuffer(); const decoder = new TextDecoder('iso-8859-1'); const textoPend = decoder.decode(bufferPend); const resPend = parsearCSV(textoPend); cabeceras = resPend.cabeceras; datosPendientes = resPend.filas; if (!cabeceras.includes('Inventario Disponible en el punto de origen')) { cabeceras.push('Inventario Disponible en el punto de origen'); } if (!cabeceras.includes('Inventario BODEGA PRINCIPAL CEDI')) { cabeceras.push('Inventario BODEGA PRINCIPAL CEDI'); } // 2. Carga del archivo de inventario renombrado const archivoInventario = 'inventario.csv'; try { const respInv = await fetch(`./${encodeURIComponent(archivoInventario)}${timestamp}`); if (respInv.ok) { const bufferInv = await respInv.arrayBuffer(); const textoInv = decoder.decode(bufferInv); const resInv = parsearCSV(textoInv); procesarCruzeInventario(resInv.filas); } } catch (eInv) { console.warn('No se pudo cargar el inventario automático:', eInv); } guardarEnLocalStorage(); poblarSelectoresFiltro(); aplicarFiltros(); } catch (err) { document.getElementById('infoContador').textContent = 'Error al cargar datos automáticos. Use los botones para subir CSV.'; console.error(err); } }
